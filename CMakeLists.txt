cmake_minimum_required(VERSION 3.24)
project(obs-comment-dock VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(BUILD_PLUGIN "Build the native OBS module" ON)
option(BUILD_QT_TESTS "Build Qt protocol tests" ON)
include(CTest)
add_library(comment-core STATIC src/core/comment.cpp src/core/filter.cpp src/core/ranker.cpp src/core/recommendation-buffer.cpp)
target_include_directories(comment-core PUBLIC src)
if(BUILD_TESTING)
  add_executable(core-tests tests/core-tests.cpp)
  target_link_libraries(core-tests PRIVATE comment-core)
  add_test(NAME core COMMAND core-tests)
endif()
if(BUILD_PLUGIN OR BUILD_QT_TESTS)
  find_package(Qt6 6.6 REQUIRED COMPONENTS Core Network Widgets)
  set(CMAKE_AUTOMOC ON)
  add_library(comment-protocol STATIC src/jev/jev-client.cpp src/youtube/youtube-chat.cpp)
  target_link_libraries(comment-protocol PUBLIC comment-core Qt6::Core)
  if(BUILD_TESTING AND BUILD_QT_TESTS)
    add_executable(protocol-tests tests/protocol-tests.cpp)
    target_link_libraries(protocol-tests PRIVATE comment-protocol)
    add_test(NAME protocol COMMAND protocol-tests)
    add_executable(jev-smoke tests/jev-smoke.cpp)
    target_link_libraries(jev-smoke PRIVATE comment-protocol Qt6::Network)
  endif()
endif()
if(BUILD_PLUGIN)
  # Prefer the official OBS development SDK. A macOS installed-app fallback is
  # available with official source headers via OBS_SOURCE_DIR.
  find_package(libobs QUIET CONFIG)
  find_package(obs-frontend-api QUIET CONFIG)
  if(NOT TARGET OBS::libobs OR NOT TARGET OBS::obs-frontend-api)
    if(NOT APPLE OR NOT OBS_SOURCE_DIR)
      message(FATAL_ERROR "Provide the OBS SDK (CMAKE_PREFIX_PATH), or on macOS OBS_SOURCE_DIR and OBS_APP")
    endif()
    set(OBS_APP "/Applications/OBS.app" CACHE PATH "OBS Studio installation")
    add_library(OBS::libobs SHARED IMPORTED)
    set_target_properties(OBS::libobs PROPERTIES IMPORTED_LOCATION "${OBS_APP}/Contents/Frameworks/libobs.framework/libobs" INTERFACE_INCLUDE_DIRECTORIES "${OBS_SOURCE_DIR}/libobs;${CMAKE_CURRENT_BINARY_DIR}/obs-config")
    add_library(OBS::obs-frontend-api SHARED IMPORTED)
    set_target_properties(OBS::obs-frontend-api PROPERTIES IMPORTED_LOCATION "${OBS_APP}/Contents/Frameworks/libobs-frontend-api.1.dylib" INTERFACE_INCLUDE_DIRECTORIES "${OBS_SOURCE_DIR}/frontend/api")
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/obs-config")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/obs-config/obsconfig.h" "#pragma once\n#define OBS_DATA_PATH \"\"\n#define OBS_INSTALL_PREFIX \"\"\n#define OBS_PLUGIN_DESTINATION \"\"\n#define OBS_RELEASE_CANDIDATE 0\n#define OBS_BETA 0\n")
  endif()
  add_library(obs-comment-dock MODULE src/plugin/plugin-main.cpp src/plugin/engine.cpp src/plugin/engine.hpp src/ui/comment-dock.cpp src/ui/comment-dock.hpp)
  target_link_libraries(obs-comment-dock PRIVATE comment-protocol Qt6::Widgets Qt6::Network OBS::libobs OBS::obs-frontend-api)
  set_target_properties(obs-comment-dock PROPERTIES PREFIX "")
  if(BUILD_TESTING)
    add_executable(ui-tests tests/ui-tests.cpp src/plugin/engine.cpp src/plugin/engine.hpp src/ui/comment-dock.cpp src/ui/comment-dock.hpp)
    target_link_libraries(ui-tests PRIVATE comment-protocol Qt6::Widgets Qt6::Network)
    add_test(NAME ui COMMAND ui-tests -platform offscreen)
  endif()

  if(APPLE)
    set_target_properties(obs-comment-dock PROPERTIES BUNDLE TRUE BUNDLE_EXTENSION plugin MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/src/plugin/Info.plist.in" BUILD_RPATH "${OBS_APP}/Contents/Frameworks" INSTALL_RPATH "@executable_path/../Frameworks")
    install(TARGETS obs-comment-dock LIBRARY DESTINATION .)
  elseif(WIN32)
    install(TARGETS obs-comment-dock RUNTIME DESTINATION obs-plugins/64bit LIBRARY DESTINATION obs-plugins/64bit)
  else()
    install(TARGETS obs-comment-dock LIBRARY DESTINATION lib/obs-plugins)
  endif()
endif()
